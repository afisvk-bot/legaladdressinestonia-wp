(function () {
  var NAME = 'lae_cookie_consent';

  function gtag(){ window.dataLayer = window.dataLayer || []; window.dataLayer.push(arguments); }

  function saveCookie(value) {
    var d = new Date();
    d.setTime(d.getTime() + 365 * 24 * 60 * 60 * 1000);
    document.cookie = NAME + '=' + value + '; expires=' + d.toUTCString() + '; path=/; SameSite=Lax; Secure';
  }

  function deleteGaCookies() {
    var host = location.hostname.replace(/^www\./, '');
    document.cookie.split(';').forEach(function (c) {
      var name = c.split('=')[0].trim();
      if (/^(_ga|_gid|_gat)/.test(name)) {
        ['', host, '.' + host].forEach(function (domain) {
          document.cookie = name + '=; expires=Thu, 01 Jan 1970 00:00:00 GMT; path=/' + (domain ? '; domain=' + domain : '');
        });
      }
    });
  }

  function choose(value) {
    saveCookie(value);
    var banner = document.getElementById('lae-cookie-consent');
    if (banner) banner.setAttribute('hidden', '');

    if (value === 'analytics_granted') {
      gtag('consent', 'update', { analytics_storage: 'granted' });
    } else {
      gtag('consent', 'update', { analytics_storage: 'denied' });
      deleteGaCookies();
    }
    window.dataLayer.push({ event: 'lae_consent_update', lae_consent: value });
  }

  document.addEventListener('click', function (e) {
    if (!e.target.closest) return;
    var el = e.target.closest('#lae-cookie-accept, #lae-cookie-reject, #lae-cookie-close, [data-lae-cookie-settings]');
    if (!el) return;

    // «Cookie settings» в футере: сбросить выбор и показать баннер заново (на языке страницы)
    if (el.hasAttribute('data-lae-cookie-settings')) {
      e.preventDefault();
      document.cookie = NAME + '=; expires=Thu, 01 Jan 1970 00:00:00 GMT; path=/';
      location.reload();
      return;
    }

    choose(el.id === 'lae-cookie-accept' ? 'analytics_granted' : 'necessary_only');
  }, true);
})();
</script>
