===== #12 LAI - UTM persistence between pages scope=front-end
/** Preserve campaign attribution during a consented visit. Disable this snippet to roll back. */
add_action('wp_head', function () {
    if (is_admin()) {
        return;
    }
    ?>
<script id="lai-utm-persistence" data-no-optimize="1" data-no-defer="1">
(function () {
    'use strict';
    const keys = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_term', 'utm_content'];
    const storageKey = 'lai_utm_visit_v1';
    const lifetime = 30 * 60 * 1000;

    function forget() {
        try { sessionStorage.removeItem(storageKey); } catch (_) {}
    }

    function preserve() {
        if (!/(?:^|;\s*)lae_cookie_consent=analytics_granted(?:;|$)/.test(document.cookie)) {
            forget();
            return;
        }
        const url = new URL(window.location.href);
        const now = Date.now();
        let values = {};
        if (keys.some(key => url.searchParams.has(key))) {
            // A new campaign replaces the whole set, never mixing old and new tags.
            for (const key of keys) {
                const value = url.searchParams.get(key);
                if (value) values[key] = value;
            }
        } else {
            let saved;
            try { saved = JSON.parse(sessionStorage.getItem(storageKey)); } catch (_) {}
            if (!saved || !Number.isFinite(saved.seenAt) || now < saved.seenAt
                || now - saved.seenAt >= lifetime || !saved.values
                || typeof saved.values !== 'object' || Array.isArray(saved.values)) {
                forget();
                return;
            }
            for (const key of keys) {
                if (typeof saved.values[key] === 'string' && saved.values[key]) {
                    values[key] = saved.values[key];
                    url.searchParams.set(key, values[key]);
                }
            }
            // The native Pipedrive loader reads document.URL when it creates the iframe.
            // Restore before that happens, without navigating or adding a history entry.
            if (Object.keys(values).length) {
                try { window.history.replaceState(window.history.state, '', url.href); } catch (_) {}
            }
        }
        if (!Object.keys(values).length) {
            forget();
            return;
        }
        try { sessionStorage.setItem(storageKey, JSON.stringify({ seenAt: now, values: values })); } catch (_) {}
    }

    preserve();
    // The existing consent handler writes its cookie in the capture phase.
    document.addEventListener('click', function (event) {
        const target = event.target;
        if (target && typeof target.closest === 'function'
            && target.closest('#lae-cookie-accept, #lae-cookie-reject, #lae-cookie-close')) {
            preserve();
        }
    });
})();
</script>
    <?php
}, -1);
